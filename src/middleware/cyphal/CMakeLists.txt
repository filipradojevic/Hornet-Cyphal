set(ADDITIONAL_SRCS)

# Run Cyphal-related generators after MAVLink and ULog are generated
# (ULog component enforces ordering via REQUIRES above)
find_program(PYTHON_EXECUTABLE NAMES python3 python REQUIRED)

set(CYPHAL_SCRIPTS_DIR "${CMAKE_CURRENT_LIST_DIR}")

set(GEN_PUBLISHERS_SCRIPT "${CYPHAL_SCRIPTS_DIR}/gen_publishers.py")
set(MAVLINK_TO_DSDL_SCRIPT "${CMAKE_SOURCE_DIR}/src/libs/libcanard/mavlink_to_dsdl.py")
set(NUNAVUT_SUPPORT_GEN TRUE)

# Paths matching LPC-style generation outputs
set(CYPHAL_GEN_ROOT "${CMAKE_BINARY_DIR}/cyphal/cyphal/messages/mavlink/messages_cyphal_uavcan")
set(CYPHAL_UTILITY_DIR "${CMAKE_BINARY_DIR}/cyphal/utility")
set(CYPHAL_MAV_PUB_SRC "${CYPHAL_UTILITY_DIR}/cyphal_mavlink_publishers.c")
set(CYPHAL_UAVCAN_DIR "${CMAKE_BINARY_DIR}/cyphal/uavcan")
set(CYPHAL_REG_DIR "${CMAKE_BINARY_DIR}/cyphal/reg")
set(CYPHAL_GEN_COMMON "${CYPHAL_GEN_ROOT}/common")
set(CYPHAL_GEN_LISUM  "${CYPHAL_GEN_ROOT}/lisum")

# Input MAVLink header roots discovered/generated by MAVLink C generator
set(MAVLINK_COMMON_DIR "${CMAKE_BINARY_DIR}/mavlink/common")
set(MAVLINK_LISUM_DIR  "${CMAKE_BINARY_DIR}/mavlink/lisum")

# Generate regulated UAVCAN and reg headers via Nunavut
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/messages/cyphal/uavcan")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/messages/cyphal/reg")
file(MAKE_DIRECTORY "${CYPHAL_UAVCAN_DIR}")
file(MAKE_DIRECTORY "${CYPHAL_REG_DIR}")
execute_process(
    COMMAND "${PYTHON_EXECUTABLE}" -m nunavut --target-language c --outdir "${CYPHAL_UAVCAN_DIR}" --allow-unregulated-fixed-port-id "${CMAKE_SOURCE_DIR}/messages/cyphal/uavcan"
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    RESULT_VARIABLE NN_UAVCAN_RETVAL
    OUTPUT_VARIABLE NN_UAVCAN_OUT
    ERROR_VARIABLE  NN_UAVCAN_ERR
)
if (NOT NN_UAVCAN_RETVAL EQUAL 0)
    message(FATAL_ERROR "Nunavut UAVCAN generation failed (rc=${NN_UAVCAN_RETVAL}):\n${NN_UAVCAN_ERR}")
endif()

execute_process(
    COMMAND "${PYTHON_EXECUTABLE}" -m nunavut --target-language c --outdir "${CYPHAL_REG_DIR}" --lookup-dir "${CMAKE_SOURCE_DIR}/messages/cyphal/uavcan" --allow-unregulated-fixed-port-id "${CMAKE_SOURCE_DIR}/messages/cyphal/reg"
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    RESULT_VARIABLE NN_REG_RETVAL
    OUTPUT_VARIABLE NN_REG_OUT
    ERROR_VARIABLE  NN_REG_ERR
)
if (NOT NN_REG_RETVAL EQUAL 0)
    message(FATAL_ERROR "Nunavut REG generation failed (rc=${NN_REG_RETVAL}):\n${NN_REG_ERR}")
endif()

if (EXISTS "${GEN_PUBLISHERS_SCRIPT}")
    file(MAKE_DIRECTORY "${CYPHAL_UTILITY_DIR}")
    # Ensure MAVLink DSDL exists (common + lisum) before publisher generation
    if (EXISTS "${MAVLINK_TO_DSDL_SCRIPT}")
        file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/cyphal/cyphal/messages/mavlink")
        if (NOT EXISTS "${CYPHAL_GEN_COMMON}" OR NOT EXISTS "${CYPHAL_GEN_LISUM}")
            execute_process(
                COMMAND "${PYTHON_EXECUTABLE}" "${MAVLINK_TO_DSDL_SCRIPT}" --mavlink-dirs "${MAVLINK_COMMON_DIR}" "${MAVLINK_LISUM_DIR}" --out-dir "${CMAKE_BINARY_DIR}/cyphal/cyphal/messages/mavlink"
                WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
                RESULT_VARIABLE CYPHAL_MAV2DSDL_PRE_RET
                OUTPUT_VARIABLE CYPHAL_MAV2DSDL_PRE_OUT
                ERROR_VARIABLE  CYPHAL_MAV2DSDL_PRE_ERR
            )
            if (NOT CYPHAL_MAV2DSDL_PRE_RET EQUAL 0)
                message(FATAL_ERROR "mavlink_to_dsdl.py (pre) failed (rc=${CYPHAL_MAV2DSDL_PRE_RET}):\n${CYPHAL_MAV2DSDL_PRE_ERR}")
            endif()
        endif()
    endif()
    # Generate MAVLink-derived publishers
    execute_process(
        COMMAND "${PYTHON_EXECUTABLE}" "${GEN_PUBLISHERS_SCRIPT}" --root "${CYPHAL_GEN_COMMON}" --root "${CYPHAL_GEN_LISUM}" --out "${CYPHAL_MAV_PUB_SRC}"
        WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
        RESULT_VARIABLE CYPHAL_GEN_PUB_RETVAL
        OUTPUT_VARIABLE CYPHAL_GEN_PUB_OUT
        ERROR_VARIABLE  CYPHAL_GEN_PUB_ERR
    )
    if (NOT CYPHAL_GEN_PUB_RETVAL EQUAL 0)
        message(FATAL_ERROR "gen_publishers.py (MAVLink) failed (rc=${CYPHAL_GEN_PUB_RETVAL}):\n${CYPHAL_GEN_PUB_ERR}")
    endif()

    # Generate regulated UAVCAN publishers
    set(CYPHAL_UAVCAN_PUB_SRC "${CMAKE_BINARY_DIR}/cyphal/utility/cyphal_uavcan_publishers.c")
    execute_process(
        COMMAND "${PYTHON_EXECUTABLE}" "${GEN_PUBLISHERS_SCRIPT}" --root "${CMAKE_SOURCE_DIR}/messages/cyphal/uavcan" --out "${CYPHAL_UAVCAN_PUB_SRC}"
        WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
        RESULT_VARIABLE CYPHAL_GEN_PUB_RETVAL2
        OUTPUT_VARIABLE CYPHAL_GEN_PUB_OUT2
        ERROR_VARIABLE  CYPHAL_GEN_PUB_ERR2
    )
    if (NOT CYPHAL_GEN_PUB_RETVAL2 EQUAL 0)
        message(FATAL_ERROR "gen_publishers.py (UAVCAN) failed (rc=${CYPHAL_GEN_PUB_RETVAL2}):\n${CYPHAL_GEN_PUB_ERR2}")
    endif()

    if (EXISTS "${CYPHAL_MAV_PUB_SRC}")
        list(APPEND ADDITIONAL_SRCS "cyphal_utility.c" "${CYPHAL_MAV_PUB_SRC}")
    endif()
    if (EXISTS "${CYPHAL_UAVCAN_PUB_SRC}")
        list(APPEND ADDITIONAL_SRCS "${CYPHAL_UAVCAN_PUB_SRC}")
    endif()
        if(CYPHAL_ULOG_GEN_SRCS)
            # Append discovered sources to this component
            target_sources(${COMPONENT_LIB} PRIVATE ${CYPHAL_ULOG_GEN_SRCS})
        endif()
endif()

if (EXISTS "${MAVLINK_TO_DSDL_SCRIPT}")
    # Convert MAVLink headers (common + lisum) to DSDL into build/cyphal tree
    set(MAVLINK_COMMON_DIR "${CMAKE_BINARY_DIR}/mavlink/common")
    set(MAVLINK_LISUM_DIR "${CMAKE_BINARY_DIR}/mavlink/lisum")
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/cyphal/cyphal/messages/mavlink")
    execute_process(
        COMMAND "${PYTHON_EXECUTABLE}" "${MAVLINK_TO_DSDL_SCRIPT}" --mavlink-dirs "${MAVLINK_COMMON_DIR}" "${MAVLINK_LISUM_DIR}" --out-dir "${CMAKE_BINARY_DIR}/cyphal/cyphal/messages/mavlink"
        WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
        RESULT_VARIABLE CYPHAL_MAV2DSDL_RETVAL
        OUTPUT_VARIABLE CYPHAL_MAV2DSDL_OUT
        ERROR_VARIABLE  CYPHAL_MAV2DSDL_ERR
    )
    if (NOT CYPHAL_MAV2DSDL_RETVAL EQUAL 0)
        message(FATAL_ERROR "mavlink_to_dsdl.py failed (rc=${CYPHAL_MAV2DSDL_RETVAL}):\n${CYPHAL_MAV2DSDL_ERR}")
    endif()

    # Generate MAVLink-derived C headers from the produced DSDL using Nunavut
    set(CYPHAL_MAV_C_DIR "${CMAKE_BINARY_DIR}/cyphal/mavlink")
    file(MAKE_DIRECTORY "${CYPHAL_MAV_C_DIR}")
    if (EXISTS "${CYPHAL_GEN_COMMON}")
        execute_process(
            COMMAND "${PYTHON_EXECUTABLE}" -m nunavut --target-language c --outdir "${CYPHAL_MAV_C_DIR}" --allow-unregulated-fixed-port-id "${CYPHAL_GEN_COMMON}"
            WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
            RESULT_VARIABLE NN_MAV_COMMON_RETVAL
            OUTPUT_VARIABLE NN_MAV_COMMON_OUT
            ERROR_VARIABLE  NN_MAV_COMMON_ERR
        )
        if (NOT NN_MAV_COMMON_RETVAL EQUAL 0)
            message(FATAL_ERROR "Nunavut MAVLink(common) generation failed (rc=${NN_MAV_COMMON_RETVAL}):\n${NN_MAV_COMMON_ERR}")
        endif()
    endif()
    if (EXISTS "${CYPHAL_GEN_LISUM}")
        execute_process(
            COMMAND "${PYTHON_EXECUTABLE}" -m nunavut --target-language c --outdir "${CYPHAL_MAV_C_DIR}" --lookup-dir "${CYPHAL_GEN_COMMON}" --allow-unregulated-fixed-port-id "${CYPHAL_GEN_LISUM}"
            WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
            RESULT_VARIABLE NN_MAV_LISUM_RETVAL
            OUTPUT_VARIABLE NN_MAV_LISUM_OUT
            ERROR_VARIABLE  NN_MAV_LISUM_ERR
        )
        if (NOT NN_MAV_LISUM_RETVAL EQUAL 0)
            message(FATAL_ERROR "Nunavut MAVLink(lisum) generation failed (rc=${NN_MAV_LISUM_RETVAL}):\n${NN_MAV_LISUM_ERR}")
        endif()
    endif()
endif()

#
# === Generate ULog ↔ Cyphal bridge code ===
#

set(CYPHAL_DSDL_ROOT "${CMAKE_SOURCE_DIR}/messages/cyphal")
set(CYPHAL_ULOG_GEN_SCRIPT "${CMAKE_SOURCE_DIR}/src/middleware/cyphal/gen_ulog_cyphal.py")
set(CYPHAL_ULOG_GEN_OUTPUT "${CMAKE_BINARY_DIR}/ulog/cyphal")

file(MAKE_DIRECTORY "${CYPHAL_ULOG_GEN_OUTPUT}")

# Paths
set(CYPHAL_ULOG_GEN_SCRIPT "${CMAKE_SOURCE_DIR}/src/middleware/cyphal/gen_ulog_cyphal.py")
set(CYPHAL_DSDL_ROOT "${CMAKE_SOURCE_DIR}/messages/cyphal")
set(CYPHAL_ULOG_GEN_OUTPUT "${CMAKE_BINARY_DIR}/ulog/cyphal")

file(MAKE_DIRECTORY "${CYPHAL_ULOG_GEN_OUTPUT}")

# ESP-IDF forbids add_custom_command during metadata scan.
# So run generator ONLY during normal configure/generate:
if(BOOTLOADER_BUILD)
    # Skip Cyphal component in bootloader build to avoid dependency resolution issues
    return()
endif()

if(NOT CMAKE_SCRIPT_MODE_FILE)

    add_custom_command(
        OUTPUT "${CYPHAL_ULOG_GEN_OUTPUT}/__generated_stamp"
        COMMAND ${CMAKE_COMMAND} -E remove_directory "${CYPHAL_ULOG_GEN_OUTPUT}"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CYPHAL_ULOG_GEN_OUTPUT}"
        COMMAND ${PYTHON_EXECUTABLE}
                "${CYPHAL_ULOG_GEN_SCRIPT}"
                "${CYPHAL_DSDL_ROOT}"
                -o "${CYPHAL_ULOG_GEN_OUTPUT}"
                --add-timestamp
        COMMAND ${CMAKE_COMMAND} -E touch "${CYPHAL_ULOG_GEN_OUTPUT}/__generated_stamp"
        DEPENDS
            "${CYPHAL_ULOG_GEN_SCRIPT}"
        COMMENT "Generating Cyphal ↔ ULog bridge code"
    )

    add_custom_target(generate_cyphal_ulog ALL
        DEPENDS "${CYPHAL_ULOG_GEN_OUTPUT}/__generated_stamp"
    )

endif()

# Finally register the component including generated sources (if any)
idf_component_register(
    SRCS
        cyphal_utility.c
        ${ADDITIONAL_SRCS}
    INCLUDE_DIRS
        .
        "${CMAKE_BINARY_DIR}/cyphal/utility"
        "${CMAKE_BINARY_DIR}/cyphal/uavcan"
        "${CMAKE_BINARY_DIR}/cyphal/reg"
        "${CMAKE_BINARY_DIR}/cyphal/mavlink"
        "${CYPHAL_ULOG_GEN_OUTPUT}"
    PRIV_REQUIRES
        esp_timer
        driver
        lfs
    REQUIRES
        libcanard
        ulog
        
)

# Ensure the component waits for ULog codegen in normal configure mode
if(NOT CMAKE_SCRIPT_MODE_FILE)
    add_dependencies(${COMPONENT_LIB} generate_cyphal_ulog)
endif()

# Pick up generated ULog sources so they build and appear in compile_commands.json
file(GLOB_RECURSE CYPHAL_ULOG_GEN_SRCS CONFIGURE_DEPENDS
    "${CMAKE_BINARY_DIR}/ulog/cyphal/*.c")
if(CYPHAL_ULOG_GEN_SRCS)
    target_sources(${COMPONENT_LIB} PRIVATE ${CYPHAL_ULOG_GEN_SRCS})
endif()
